        using System;
using System.Security.Cryptography;

namespace lab_6
{
    class Program
    {
        static void Main(string[] args)
        {
            DSACryptoServiceProvider myDSA = new DSACryptoServiceProvider(1024); // Changed key size to 1024
            System.Diagnostics.Stopwatch swatch = new System.Diagnostics.Stopwatch();
            int count = 100;

            // Generate a DSA key
            swatch.Start();
            for (int i = 0; i < count; i++)
            {
                myDSA = new DSACryptoServiceProvider(1024); // Changed key size to 1024
            }
            swatch.Stop();
            Console.WriteLine("Key generation time at 1024 bit ... " + (swatch.ElapsedTicks / (10 * count)).ToString() + " ms");

            // Message to sign
            string message = "This is an important message";
            byte[] messageBytes = System.Text.Encoding.ASCII.GetBytes(message);

            // Sign the message
            swatch.Reset();
            swatch.Start();
            byte[] signature = null;
            for (int i = 0; i < count; i++)
            {
                signature = myDSA.SignData(messageBytes);
            }
            swatch.Stop();
            Console.WriteLine("Signing time at 1024 bit ... " + (swatch.ElapsedTicks / (10 * count)).ToString() + " ms");

            // Verify the signature
            swatch.Reset();
            swatch.Start();
            bool verified = false;
            for (int i = 0; i < count; i++)
            {
                verified = myDSA.VerifyData(messageBytes, signature);
            }
            swatch.Stop();
            Console.WriteLine("Verification time at 1024 bit ... " + (swatch.ElapsedTicks / (10 * count)).ToString() + " ms");

            Console.WriteLine("Signature verified: " + verified);
        }
    }
}

Java BIG number

        import java.math.BigInteger;

public class App {
    public static void main(String[] args) throws Exception {
        BigInteger n= new BigInteger("837210799");
        BigInteger e= new BigInteger("7");
        BigInteger d= new BigInteger("478341751");

        BigInteger one= BigInteger.ONE;
        BigInteger k;
        k=d.multiply(e).subtract(one).divide(n).add(one);
        BigInteger p_plus_q;
        p_plus_q= k.multiply(n.add(one)).add(one.subtract(e.multiply(d))).divide(k);
        System.out.println("p+q: "+p_plus_q.toString());
        BigInteger delta;
        delta= p_plus_q.multiply(p_plus_q).subtract(k.multiply(n));
        BigInteger two= new BigInteger("2");
        BigInteger x1= p_plus_q.add(delta.sqrt()).divide(two);
        System.out.println("delta: "+delta.toString());
        System.out.println("x1: "+x1.toString());
        BigInteger x2= p_plus_q.subtract(delta.sqrt()).divide(two);
        System.out.println("x2: "+x2.toString());
    }
}

Java Cryptography Example

        import java.security.Key; 
        import java.security.KeyPair; 
        import java.security.KeyPairGenerator; 
        import java.security.SecureRandom; 
        import javax.crypto.Cipher;
        import javax.crypto.SecretKeyFactory;
        import javax.crypto.SecretKey;
        import java.math.BigInteger;
        import javax.crypto.spec.PBEKeySpec;
        import javax.crypto.spec.SecretKeySpec; 

        public class App {
            public static void main(String[] args) throws Exception {
                byte[] keyBytes = new byte[16];
                SecureRandom myPRNG = new SecureRandom();

                char[] password = "short_password".toCharArray();
                byte[] salt = new byte[16]; 
                int iteration_count = 10000; 
                int key_length = 128;

                myPRNG.nextBytes(salt);
                SecretKeyFactory keyFactory = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA1");
                PBEKeySpec pbecSpec = new PBEKeySpec(password, salt, iteration_count, key_length);
                SecretKey myAESPBKey = new SecretKeySpec(keyFactory.generateSecret(pbecSpec).getEncoded(), "AES");
                keyBytes = myAESPBKey.getEncoded();
                System.out.println("AES key: " + new BigInteger(myAESPBKey.getEncoded()).toString(16));

                SecretKeySpec myKey = new SecretKeySpec(keyBytes, "AES");
                Cipher myAES = Cipher.getInstance("AES/ECB/NoPadding");

                byte[] plaintext = new byte[16];
                byte[] ciphertext = new byte[16];
                myAES.init(Cipher.ENCRYPT_MODE, myKey);
                int cLength = myAES.update(plaintext, 0, plaintext.length, ciphertext, 0);
                cLength += myAES.doFinal(ciphertext, cLength);

                System.out.println("plaintext: " + new BigInteger(plaintext).toString(16));
                System.out.println("ciphertext: " + new BigInteger(ciphertext).toString(16));

                myAES.init(Cipher.DECRYPT_MODE, myKey);
                byte[] dec_plaintext = new byte[16];
                cLength = myAES.update(ciphertext, 0, ciphertext.length, dec_plaintext, 0);
                cLength += myAES.doFinal(dec_plaintext, cLength);

                System.out.println("decrypted: " + new BigInteger(dec_plaintext).toString(16));

                KeyPairGenerator keyGen = KeyPairGenerator.getInstance("RSA");
                keyGen.initialize(2048, myPRNG);
                KeyPair myRSAKeyPair = keyGen.generateKeyPair();

                Cipher myRSA = Cipher.getInstance("RSA/ECB/PKCS1Padding");
                Key pbKey = myRSAKeyPair.getPublic();
                Key pvKey = myRSAKeyPair.getPrivate();

                myRSA.init(Cipher.ENCRYPT_MODE, pbKey);
                ciphertext = myRSA.doFinal(keyBytes);

                myRSA.init(Cipher.DECRYPT_MODE, pvKey);
                plaintext = myRSA.doFinal(ciphertext);

                System.out.println("plaintext: " + new BigInteger(plaintext).toString(16));
                System.out.println("ciphertext: " + new BigInteger(ciphertext).toString(16));
                System.out.println("keybytes: " + new BigInteger(keyBytes).toString(16));

                String plain = "ABCDEFGH12345678"; // 16 bytes
                plaintext = plain.getBytes();
            }
        }
